// Di dalam guru.html, tambahkan variabel global untuk menyimpan status izin hapus let izinHapusPenugasan = true; // default true // Di dalam window.onload, ambil pengaturan dari Firebase window.onload = async () => { const guruLogin = localStorage.getItem('guruLogin'); if (!guruLogin) { window.location.href = 'index.html'; return; } currentTeacher = guruLogin; document.getElementById('greeting').innerHTML = ` ${escapeHtml(currentTeacher)} · Guru Aktif`; // Ambil pengaturan izin hapus try { const snap = await db.ref('settings/guru/hapusPenugasan').once('value'); const val = snap.val(); izinHapusPenugasan = (val === undefined || val === null) ? true : val; } catch (e) { console.warn('Gagal memuat pengaturan izin hapus, menggunakan default: true', e); izinHapusPenugasan = true; } await fetchInitialData(); await renderAllCards(); document.getElementById('backToListBtn').onclick = backToList; const logoutBtn = document.getElementById('logoutBtn'); if (logoutBtn) { logoutBtn.addEventListener('click', (e) => { e.preventDefault(); localStorage.removeItem('guruLogin'); window.location.href = 'index.html'; }); } }; // Di dalam renderAllCards, sesuaikan tampilan tombol hapus berdasarkan izin async function renderAllCards() { const container = document.getElementById('assignmentsContainer'); if (!penugasanList.length) { container.innerHTML = `

Belum ada penugasan. Klik "+ Penugasan" untuk memulai.

`; return; } let html = ''; for (let task of penugasanList) { let inactiveCount = 0; for (let nis of task.siswaList) { if (allSiswa[nis] && allSiswa[nis].status === 'tidak_aktif') inactiveCount++; } const inactiveBadge = inactiveCount > 0 ? `${inactiveCount} tidak aktif` : ''; // Tampilkan tombol hapus hanya jika izinHapusPenugasan = true const deleteButton = izinHapusPenugasan ? `` : ''; html += `

${escapeHtml(task.mapelNama)} ${inactiveBadge}

Kelas: ${escapeHtml(task.kelasDisplay)}

${task.mapelJenis === 'inkul' ? 'Intrakurikuler' : (task.mapelJenis === 'eskul' ? 'Ekskul' : 'Pesantren')}

${deleteButton}
`; } container.innerHTML = html; // Event listener untuk klik card container.querySelectorAll('.card-assignment').forEach(card => { card.addEventListener('click', (e) => { if (e.target.closest('.btn-delete-card')) return; const selectedTask = penugasanList.find(t => t.id === card.dataset.id); if (selectedTask) showDetailAssignment(selectedTask); }); }); // Event listener untuk tombol hapus (hanya jika izin aktif) if (izinHapusPenugasan) { container.querySelectorAll('.btn-delete-card').forEach(btn => { btn.addEventListener('click', async (e) => { e.stopPropagation(); const id = btn.dataset.id; await deleteAssignment(id); }); }); } }
